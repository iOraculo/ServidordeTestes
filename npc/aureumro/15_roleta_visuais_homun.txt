// =============================================================================
// AureumRO Servidor de Testes - Roleta Poring, Visuais (Costumes), Auras & Homun
// Itens 2, 8 e 9:
//   - Roleta Poring interativa estilo AureumRO
//   - Armário de Visuais (aba Costume: Topo/Meio/Baixo/Capa) sem ocupar equip +10
//   - Auras Contínuas persistentes (reaplicadas via timer e ao logar)
//   - Assistente de Homúnculos (Embrião, Lealdade 1000, Evolução e Nível 99)
// =============================================================================

prontera,166,195,3	script	Roleta & Visuais	1002,{
	mes "[ ^B8860BRoleta Poring, Visuais & Auras^000000 ]";
	mes "Explore os sistemas temáticos e visuais do ^B8860BAureumRO^000000!";
	next;
	switch (select(
		"^FF6699Girar a Roleta Poring (Grátis em Testes)^000000",
		"^B8860BArmário de Visuais - Aba Costume (Topo/Meio/Baixo/Capa)^000000",
		"^0055FFAuras Contínuas (Ligar / Trocar / Desligar)^000000",
		"Chapéus Clássicos Pré-Renewal",
		"^006600Homúnculos Express (Alquimista / Criador)^000000",
		"Cancelar"
	)) {
	case 1:
		callsub L_PoringRoulette;
		close;
	case 2:
		callsub L_CostumeCloset;
		close;
	case 3:
		callsub L_ContinuousAuras;
		close;
	case 4:
		callsub L_ClassicHats;
		close;
	case 5:
		callsub L_HomunculusExpress;
		close;
	case 6:
		close;
	}

L_PoringRoulette:
	mes "[ ^FF6699Roleta Poring AureumRO^000000 ]";
	mes "Girando a roleta... boa sorte!";
	specialeffect2 EF_COIN;
	next;
	.@roll = rand(1, 100);
	if (.@roll <= 5) {
		getitem 616, 1;
		specialeffect2 EF_ANGEL2;
		mes "^B8860B[JACKPOT!]^000000 Você ganhou ^0055FF1x Velho Álbum de Cartas (OCA)^000000!";
	} else if (.@roll <= 20) {
		getitem 607, 25;
		mes "^006600[PRÊMIO RARO]^000000 Você ganhou ^0055FF25x Fruto de Yggdrasil^000000!";
	} else if (.@roll <= 40) {
		getitem 23044, 20;
		mes "^006600[PRÊMIO]^000000 Você ganhou ^B8860B20x Doce de Elvira^000000!";
	} else if (.@roll <= 65) {
		getitem 603, 5;
		mes "[PRÊMIO] Você ganhou ^0055FF5x Velha Caixa Azul (OBB)^000000!";
	} else if (.@roll <= 85) {
		getitem 12208, 3;
		getitem 12210, 3;
		mes "[PRÊMIO] Você ganhou ^0055FF3x Manual de Combate + 3x Goma de Mascar^000000!";
	} else {
		Zeny += 500000;
		mes "[PRÊMIO] Você ganhou ^B8860B500.000 Zeny^000000!";
	}
	return;

L_CostumeCloset:
	mes "[ ^B8860BArmário de Visuais (Aba Costume)^000000 ]";
	mes "Estes itens equipam na aba ^0055FFVisual (Costume)^000000 sem remover seus equipamentos Divinos +10!";
	next;
	.@pick = select(
		"[Topo] Faixa de Penas da Valquíria (19506)",
		"[Topo] Sol Radiante / Clear Sun (19507)",
		"[Topo] Elmo da Esfinge Visual (19501)",
		"[Topo] Pintinho Visual / Baby Chick (19608)",
		"[Topo] Boina Tiraya Visual (19655)",
		"[Meio] Orelhas de Asa de Borboleta (19509)",
		"[Meio] Senhor dos Mortos Visual (19574)",
		"[Baixo] Bigode Visual (19514)",
		"[Capa] Asas do Cupido Visual (20504)",
		"[Capa] Mochila Poring Visual (20507)",
		"Cancelar"
	);
	if (.@pick == 11) return;
	setarray .@cids[1], 19506, 19507, 19501, 19608, 19655, 19509, 19574, 19514, 20504, 20507;
	getitem .@cids[.@pick], 1;
	specialeffect2 EF_BEGINSPELL6;
	mes "Visual entregue! Equipe-o pela aba Visual do inventário (Alt+E).";
	return;

L_ContinuousAuras:
	mes "[ ^0055FFAuras Contínuas AureumRO^000000 ]";
	mes "Aura atual: " + (AUREUM_AURA_ID ? "^006600Ativa (#" + AUREUM_AURA_ID + ")^000000" : "^777777Desligada^000000");
	mes "Escolha uma aura que acompanha continuamente seu personagem:";
	next;
	.@opt = select(
		"Aura Dourada Real (Ouro 99)",
		"Aura Sagrada (Brilho Assumptio)",
		"Aura de Chamas (Espírito da Fúria)",
		"Aura Celestial (Luz Angelical)",
		"Desligar Aura Contínua",
		"Cancelar"
	);
	if (.@opt == 6) return;
	if (.@opt == 5) {
		AUREUM_AURA_ID = 0;
		deltimer strnpcinfo(3) + "::OnAuraTick";
		mes "Aura contínua desligada.";
		return;
	}
	AUREUM_AURA_ID = .@opt;
	deltimer strnpcinfo(3) + "::OnAuraTick";
	callsub L_PlayAuraEffect, AUREUM_AURA_ID;
	addtimer 4500, strnpcinfo(3) + "::OnAuraTick";
	mes "Aura contínua ativada!";
	return;

L_PlayAuraEffect:
	switch (getarg(0)) {
	case 1: specialeffect2 EF_LEVEL99_4; break;
	case 2: specialeffect2 EF_ASSUMPTIO; break;
	case 3: specialeffect2 EF_EXPLOSIONSPIRITS; break;
	case 4: specialeffect2 EF_ANGEL2; break;
	}
	return;

OnAuraTick:
	if (AUREUM_AURA_ID > 0) {
		callsub L_PlayAuraEffect, AUREUM_AURA_ID;
		addtimer 4500, strnpcinfo(3) + "::OnAuraTick";
	}
	end;

OnPCLoginEvent:
	if (AUREUM_AURA_ID > 0) {
		addtimer 2500, strnpcinfo(3) + "::OnAuraTick";
	}
	end;

L_ClassicHats:
	mes "[ ^B8860BChapéus Clássicos Pré-Renewal^000000 ]";
	next;
	.@pick = select(
		"Boina Alada (5170)",
		"Chifres Majestosos Gigantes (5374)",
		"Boneca de Alice (5137)",
		"Chapéu de Rideword (5208)",
		"Chapéu de Capitão (5359)",
		"Orelhas de Gato Preto (5360)",
		"Tiara de PecoPeco (5358)",
		"Olho Biônico (5325)",
		"Espírito do Anjo (5389)",
		"Orelhas de Elfo (2286)",
		"Cancelar"
	);
	if (.@pick == 11) return;
	setarray .@vids[1], 5170, 5374, 5137, 5208, 5359, 5360, 5358, 5325, 5389, 2286;
	getitem .@vids[.@pick], 1;
	specialeffect2 EF_BEGINSPELL6;
	mes "Chapéu entregue!";
	return;

L_HomunculusExpress:
	mes "[ ^006600Homúnculos Express (Alquimista / Criador)^000000 ]";
	if (BaseJob != Job_Alchemist) {
		mes "Dica: você não é Alquimista/Criador no momento, mas ainda pode pegar Embriões.";
	}
	next;
	switch (select(
		"Receber 10x Embrião (7142) + Bioética",
		"Maximizar Lealdade do Homúnculo (1000 - Alta)",
		"Evoluir Homúnculo imediatamente (homevolution)",
		"Subir Homúnculo para Nível 99 (@homlvl 99)",
		"Alimentar e Curar Homúnculo",
		"Cancelar"
	)) {
	case 1:
		getitem 7142, 10;
		if (BaseJob == Job_Alchemist) {
			skill "AM_BIOETHICS", 1, SKILL_PERM;
			skill "AM_CALLHOMUN", 1, SKILL_PERM;
			skill "AM_REST", 1, SKILL_PERM;
			skill "AM_RESURRECTHOMUN", 5, SKILL_PERM;
		}
		mes "10x Embrião entregues! Use a skill 'Criar Homúnculo' para invocar.";
		return;
	case 2:
		if (!gethominfo(0)) { mes "Invoque seu Homúnculo primeiro!"; return; }
		atcommand "@homfriendly 1000";
		specialeffect2 EF_HEARTCASTING;
		mes "Lealdade do Homúnculo definida para 1000 (Alta)!";
		return;
	case 3:
		if (!gethominfo(0)) { mes "Invoque seu Homúnculo primeiro!"; return; }
		atcommand "@homfriendly 1000";
		homevolution;
		specialeffect2 EF_ANGEL2;
		mes "Evolução do Homúnculo aplicada!";
		return;
	case 4:
		if (!gethominfo(0)) { mes "Invoque seu Homúnculo primeiro!"; return; }
		atcommand "@homlvl 99";
		specialeffect2 EF_ANGEL2;
		mes "Homúnculo elevado para o Nível 99!";
		return;
	case 5:
		if (!gethominfo(0)) { mes "Invoque seu Homúnculo primeiro!"; return; }
		atcommand "@homhungry 100";
		percentheal 100, 100;
		mes "Fome do Homúnculo restaurada para 100%!";
		return;
	default:
		return;
	}

OnInit:
	waitingroom "[ Roleta & Visuais ]", 0;
	end;
}
