// =============================================================================
// AureumRO Servidor de Testes - Utilidades QoL, Lixeira & Comandos @ Custom
// Itens 11 e 12:
//   - Kafra Universal + Lixeira / Limpador de Inventário + Reparador/Identificador
//   - Comandos globais via bindatcmd: @restock, @buff, @arena, @hub
// =============================================================================

prontera,164,189,3	script	Kafra & Lixeira	112,{
	mes "[ ^B8860BKafra Universal & Utilidades^000000 ]";
	mes "Como posso organizar seus itens de teste hoje?";
	next;
	switch (select(
		"Abrir Armazém (@storage)",
		"Reparar Todos os Equipamentos & Identificar Itens",
		"Reabastecer Consumíveis 4Tools (@restock)",
		"^FF0000Limpar Inventário (remover itens não-equipados)^000000",
		"Salvar Ponto de Retorno em Prontera",
		"Cancelar"
	)) {
	case 1:
		close2;
		openstorage;
		end;
	case 2:
		atcommand "@repairall";
		atcommand "@identify";
		specialeffect2 EF_REPAIRWEAPON;
		mes "Todos os equipamentos foram reparados e itens identificados!";
		close;
	case 3:
		callsub L_DoRestock;
		mes "Consumíveis 4Tools reabastecidos!";
		close;
	case 4:
		mes "[ ^FF0000Lixeira - Confirmação^000000 ]";
		mes "Isso apagará ^FF0000TODOS os itens não-equipados^000000 da sua mochila (itens equipados e no @storage permanecem intactos).";
		mes "Deseja continuar?";
		next;
		if (select("Cancelar", "Sim, limpar itens não-equipados agora") == 1) close;
		getinventorylist;
		.@removed = 0;
		for (.@i = 0; .@i < @inventorylist_count; .@i++) {
			if (@inventorylist_equip[.@i] == 0) {
				delitem @inventorylist_id[.@i], @inventorylist_amount[.@i];
				.@removed++;
			}
		}
		mes "Inventário limpo! " + .@removed + " pilha(s) de itens não-equipados removida(s).";
		close;
	case 5:
		savepoint "prontera", 156, 191;
		mes "Ponto de retorno salvo em Prontera (156, 191)!";
		close;
	case 6:
		close;
	}

L_DoRestock:
	callsub L_TopUp, 547, 150;   // Poção Branca Compacta
	callsub L_TopUp, 505, 80;    // Poção Azul
	callsub L_TopUp, 607, 40;    // Fruto de Yggdrasil
	callsub L_TopUp, 608, 40;    // Semente de Yggdrasil
	callsub L_TopUp, 506, 50;    // Poção Verde
	callsub L_TopUp, 525, 50;    // Panaceia
	callsub L_TopUp, 523, 30;    // Água Benta
	callsub L_TopUp, 23044, 30;  // Doce de Elvira
	callsub L_TopUp, 657, 15;    // Poção da Fúria Selvagem
	callsub L_TopUp, 717, 50;    // Gema Azul
	specialeffect2 EF_COIN;
	return;

L_TopUp:
	.@id = getarg(0);
	.@target = getarg(1);
	.@have = countitem(.@id);
	if (.@have < .@target) {
		getitem .@id, .@target - .@have;
	}
	return;

OnInit:
	waitingroom "[ Kafra & QoL ]", 0;
	end;
}

// Duplicata na Arena de Treino 4Tools
pvp_n_1-5,88,104,5	duplicate(Kafra & Lixeira)	Kafra & Lixeira#4tools	112

// Comandos rápidos globais de Servidor de Testes
-	script	AureumRO_AtCommands	-1,{
OnInit:
	bindatcmd "restock",    strnpcinfo(3) + "::OnRestock",    0, 99;
	bindatcmd "buff",       strnpcinfo(3) + "::OnBuff",       0, 99;
	bindatcmd "full",       strnpcinfo(3) + "::OnFull",       0, 99;
	bindatcmd "debuff",     strnpcinfo(3) + "::OnClearDebuff",0, 99;
	bindatcmd "arena",      strnpcinfo(3) + "::OnArena",      0, 99;
	bindatcmd "hub",        strnpcinfo(3) + "::OnHub",        0, 99;
	bindatcmd "mvp",        strnpcinfo(3) + "::OnMvpRoom",    0, 99;
	bindatcmd "dummy",      strnpcinfo(3) + "::OnSpawnDummy", 0, 99;
	bindatcmd "cleardummy", strnpcinfo(3) + "::OnClearDummy", 0, 99;
	bindatcmd "turbo",      strnpcinfo(3) + "::OnTurbo",      0, 99;
	bindatcmd "pacifico",   strnpcinfo(3) + "::OnPacifico",   0, 99;
	bindatcmd "kafra",      strnpcinfo(3) + "::OnKafra",      0, 99;
	bindatcmd "reparar",    strnpcinfo(3) + "::OnReparar",    0, 99;
	end;

OnRestock:
	setarray .@ids[0],   547, 505, 607, 608, 506, 525, 523, 23044, 657, 717;
	setarray .@qtys[0],  150,  80,  40,  40,  50,  50,  30,    30,  15,  50;
	for (.@i = 0; .@i < getarraysize(.@ids); .@i++) {
		.@have = countitem(.@ids[.@i]);
		if (.@have < .@qtys[.@i]) {
			getitem .@ids[.@i], .@qtys[.@i] - .@have;
		}
	}
	specialeffect2 EF_COIN;
	dispbottom "[AureumRO] @restock concluído! Poções, Yggdrasil, Panaceia, Doce de Elvira e Gemas reabastecidos.";
	end;

OnBuff:
	percentheal 100, 100;
	callsub L_RemoveDebuffs;
	sc_start SC_BLESSING, 600000, 10;
	sc_start SC_INCREASEAGI, 600000, 10;
	sc_start SC_ASSUMPTIO, 300000, 5;
	specialeffect2 EF_HEAL2;
	specialeffect2 EF_BLESSING;
	specialeffect2 EF_INCAGILITY;
	dispbottom "[AureumRO] @buff aplicado (HP/SP 100% + Bênção + Agilidade + Assumptio).";
	end;

OnFull:
	percentheal 100, 100;
	callsub L_RemoveDebuffs;
	atcommand "@repairall";
	atcommand "@identifyall";
	sc_start SC_BLESSING, 600000, 10;
	sc_start SC_INCREASEAGI, 600000, 10;
	sc_start SC_ASSUMPTIO, 600000, 5;
	sc_start SC_MAGNIFICAT, 600000, 5;
	sc_start SC_GLORIA, 600000, 5;
	sc_start SC_IMPOSITIO, 600000, 5;
	specialeffect2 EF_HEAL2;
	specialeffect2 EF_ASSUMPTIO;
	dispbottom "[AureumRO] @full aplicado! HP/SP 100%, debuffs limpos, itens reparados/identificados e buffs completos.";
	end;

OnClearDebuff:
	callsub L_RemoveDebuffs;
	specialeffect2 EF_CURE;
	dispbottom "[AureumRO] @debuff concluído! Todos os status negativos foram removidos sem apagar seus buffs.";
	end;

L_RemoveDebuffs:
	sc_end SC_POISON;
	sc_end SC_SILENCE;
	sc_end SC_BLIND;
	sc_end SC_CURSE;
	sc_end SC_CONFUSION;
	sc_end SC_BLEEDING;
	sc_end SC_STUN;
	sc_end SC_FREEZE;
	sc_end SC_STONE;
	sc_end SC_SLEEP;
	sc_end SC_HALLUCINATION;
	sc_end SC_CRITICALWOUND;
	sc_end SC_BURNING;
	sc_end SC_FREEZING;
	sc_end SC_OVERHEAT;
	sc_end SC_MANDRAGORA;
	sc_end SC_SLOWCAST;
	sc_end SC_CHANGEUNDEAD;
	sc_end SC_DECREASEAGI;
	sc_end SC_QUAGMIRE;
	return;

OnArena:
	warp "pvp_n_1-5", 100, 100;
	dispbottom "[AureumRO] Teleportado para a Arena de Treino 4Tools (pvp_n_1-5).";
	end;

OnHub:
	warp "prontera", 156, 191;
	dispbottom "[AureumRO] Teleportado para o Hub de Prontera.";
	end;

OnMvpRoom:
	warp "guild_vs3", 50, 50;
	dispbottom "[AureumRO] Teleportado para a Sala MVP Privada (guild_vs3).";
	end;

OnSpawnDummy:
	getmapxy .@map$, .@x, .@y, BL_PC;
	if (.@map$ != "pvp_n_1-5" && .@map$ != "guild_vs3") {
		dispbottom "[AureumRO] Use @dummy dentro da Arena (@arena) ou na Sala MVP (@mvp)!";
		end;
	}
	.@gid = monster(.@map$, .@x + 2, .@y, "Dummy 5M HP (DEF 0)", 1905, 1, strnpcinfo(3) + "::OnCustomDummyDead");
	if (.@gid > 0) {
		setunitdata .@gid, UMOB_MAXHP, 5000000;
		setunitdata .@gid, UMOB_HP, 5000000;
		setunitdata .@gid, UMOB_DEF, 0;
		setunitdata .@gid, UMOB_MDEF, 0;
		setunitdata .@gid, UMOB_ATKMIN, 0;
		setunitdata .@gid, UMOB_ATKMAX, 0;
		dispbottom "[AureumRO] Dummy 5M HP invocado em " + .@map$ + " (" + (.@x + 2) + ", " + .@y + "). Use @cleardummy para remover.";
	}
	end;

OnClearDummy:
	getmapxy .@map$, .@x, .@y, BL_PC;
	killmonster .@map$, strnpcinfo(3) + "::OnCustomDummyDead";
	dispbottom "[AureumRO] Todos os dummies invocados por @dummy em " + .@map$ + " foram removidos.";
	end;

OnCustomDummyDead:
	end;

OnTurbo:
	if (@aureum_turbo) {
		@aureum_turbo = 0;
		atcommand "@speed 150";
		dispbottom "[AureumRO] Modo @turbo DESATIVADO (velocidade padrão).";
	} else {
		@aureum_turbo = 1;
		atcommand "@speed 100";
		specialeffect2 EF_INCAGILITY;
		dispbottom "[AureumRO] Modo @turbo ATIVADO (movimento rápido para testes). Digite @turbo novamente para voltar ao normal.";
	}
	end;

OnPacifico:
	atcommand "@monsterignore";
	dispbottom "[AureumRO] Modo @pacifico alternado! (Monstros/MVPs ignoram você enquanto estiver ativo.)";
	end;

OnKafra:
	openstorage;
	end;

OnReparar:
	atcommand "@repairall";
	atcommand "@identifyall";
	specialeffect2 EF_REPAIRWEAPON;
	dispbottom "[AureumRO] Todos os equipamentos reparados e itens identificados!";
	end;
}
