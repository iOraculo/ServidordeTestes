// =============================================================================
// AureumRO Servidor de Testes — Mestra das Classes (Pré-Renewal 99/70)
// Suporta 1ª, 2ª, Transclasses (Reborn 99/70), Expandidas, Skills de Platina,
// Reset de Atributos/Habilidades e Atalho de Nível para Testes
// =============================================================================

prontera,150,191,5	script	Mestra das Classes	123,{
	mes "[ ^B8860BMestra das Classes^000000 ]";
	mes "Bem-vindo ao serviço de classes Pré-Renewal (Cap 99/70) do ^B8860BAureumRO^000000.";
	next;
	switch (select(
		"^0055FF[Direto] Escolher Transclasse 99/70 + Todas Skills Max^000000",
		"^0055FF[Direto] Escolher Classe Expandida 99 + Todas Skills Max^000000",
		"Maximizar Todas as Skills da Classe Atual",
		"Trocar de Classe / Reborn (Modo Progressão)",
		"Receber Habilidades de Platina",
		"Alugar Carrinho / Falcão / PecoPeco",
		"Resetar Atributos e Habilidades",
		"[Teste] Subir para Nível 99/70 imediatamente",
		"Cancelar"
	)) {
	case 1:
		callsub L_DirectTransclass;
		close;
	case 2:
		callsub L_DirectExpanded;
		close;
	case 3:
		atcommand "@allskill";
		callsub L_PlatinumSkills;
		percentheal 100, 100;
		mes "[ ^B8860BMestra das Classes^000000 ]";
		mes "Todas as habilidades da sua classe atual foram maximizadas!";
		close;
	case 4:
		callsub L_ChangeJob;
		close;
	case 5:
		callsub L_PlatinumSkills;
		close;
	case 6:
		callsub L_Rental;
		close;
	case 7:
		mes "[ ^B8860BMestra das Classes^000000 ]";
		mes "O que deseja resetar gratuitamente?";
		next;
		switch (select("Resetar Atributos", "Resetar Habilidades", "Resetar Ambos", "Cancelar")) {
		case 1:
			resetstatus;
			mes "Seus atributos foram resetados!";
			close;
		case 2:
			resetskill;
			mes "Suas habilidades foram resetadas!";
			close;
		case 3:
			resetstatus;
			resetskill;
			mes "Atributos e habilidades resetados!";
			close;
		case 4:
			close;
		}
	case 8:
		if (BaseLevel < 99) {
			BaseLevel = 99;
		}
		.@max_job = (Upper == 1 || Class == Job_SuperNovice || Class == Job_Gunslinger || Class == Job_Ninja) ? 70 : 50;
		if (Class == Job_Novice || Class == Job_Novice_High) {
			.@max_job = 10;
		}
		if (JobLevel < .@max_job) {
			JobLevel = .@max_job;
		}
		percentheal 100, 100;
		specialeffect2 EF_ANGEL2;
		mes "[ ^B8860BMestra das Classes^000000 ]";
		mes "Pronto! Seu personagem foi elevado para o nível máximo da classe atual (" + BaseLevel + "/" + JobLevel + ").";
		close;
	case 9:
		close;
	}

L_DirectTransclass:
	mes "[ ^B8860BMestra das Classes - Transclasse Direta^000000 ]";
	mes "Escolha a ^0055FFTransclasse (99/70)^000000 desejada:";
	mes "Todas as skills + Habilidades de Platina serão maximizadas automaticamente!";
	next;
	.@pick = select(
		"Lorde (Lord Knight)",
		"Paladino (Paladin)",
		"Arquimago (High Wizard)",
		"Professor (Scholar)",
		"Atirador de Elite (Sniper)",
		(Sex == SEX_MALE ? "Menestrel (Clown)" : "Cigana (Gypsy)"),
		"Sumo Sacerdote (High Priest)",
		"Mestre (Champion)",
		"Mestre-Ferreiro (Whitesmith)",
		"Criador (Creator)",
		"Algoz (Assassin Cross)",
		"Desordeiro (Stalker)",
		"Cancelar"
	);
	if (.@pick == 13) return;

	setarray .@jobs[1],
		Job_Lord_Knight,
		Job_Paladin,
		Job_High_Wizard,
		Job_Professor,
		Job_Sniper,
		(Sex == SEX_MALE ? Job_Clown : Job_Gypsy),
		Job_High_Priest,
		Job_Champion,
		Job_Whitesmith,
		Job_Creator,
		Job_Assassin_Cross,
		Job_Stalker;

	callsub L_ApplyDirectClass, .@jobs[.@pick], 70;
	return;

L_DirectExpanded:
	mes "[ ^B8860BMestra das Classes - Expandidas^000000 ]";
	mes "Escolha a ^0055FFClasse Expandida^000000 desejada (nível máximo + todas as skills maximizadas):";
	next;
	.@pick = select(
		"Super Aprendiz (99/99)",
		"Justiceiro (99/70)",
		"Ninja (99/70)",
		"Mestre Taekwon (99/50)",
		"Espiritualista (99/50)",
		"Cancelar"
	);
	if (.@pick == 6) return;

	setarray .@jobs[1], Job_SuperNovice, Job_Gunslinger, Job_Ninja, Job_Star_Gladiator, Job_Soul_Linker;
	setarray .@jlv[1], 99, 70, 70, 50, 50;

	callsub L_ApplyDirectClass, .@jobs[.@pick], .@jlv[.@pick];
	return;

L_ApplyDirectClass:
	.@target_job = getarg(0);
	.@target_jlv = getarg(1);

	if (checkcart()) setcart 0;
	if (checkfalcon()) setfalcon 0;
	if (checkriding()) setriding 0;

	resetskill;
	jobchange .@target_job;
	BaseLevel = 99;
	JobLevel = .@target_jlv;
	resetstatus;
	atcommand "@allskill";
	callsub L_PlatinumSkills;

	if (BaseClass == Job_Merchant || Class == Job_SuperNovice) {
		if (!checkcart()) setcart;
	} else if (BaseJob == Job_Hunter) {
		if (!checkfalcon()) setfalcon;
	} else if (BaseJob == Job_Knight || BaseJob == Job_Crusader) {
		if (!checkriding()) setriding;
	}

	percentheal 100, 100;
	specialeffect2 EF_ANGEL2;
	mes "[ ^B8860BMestra das Classes^000000 ]";
	mes "Pronto! Você agora é ^B8860B" + jobname(Class) + " (" + BaseLevel + "/" + JobLevel + ")^000000 com ^006600todas as skills maximizadas^000000 e atributos prontos para distribuir!";
	return;

L_ChangeJob:
	if (SkillPoint > 0) {
		mes "[ ^B8860BMestra das Classes^000000 ]";
		mes "Por favor, distribua todos os seus ^FF0000" + SkillPoint + " pontos de habilidade^000000 antes de trocar de classe.";
		return;
	}

	// Aprendiz -> 1ª Classe ou Expandida
	if (Class == Job_Novice) {
		if (JobLevel < 10) {
			mes "Você precisa de Nível de Classe 10 como Aprendiz.";
			return;
		}
		mes "Escolha sua 1ª Classe (ou Expandida):";
		next;
		.@c = select(
			"Espadachim", "Mago", "Arqueiro", "Noviço", "Mercador", "Gatuno",
			"Super Aprendiz (Base 45+)", "Taekwon", "Justiceiro", "Ninja", "Cancelar"
		);
		switch (.@c) {
		case 1: jobchange Job_Swordman; break;
		case 2: jobchange Job_Mage; break;
		case 3: jobchange Job_Archer; break;
		case 4: jobchange Job_Acolyte; break;
		case 5: jobchange Job_Merchant; break;
		case 6: jobchange Job_Thief; break;
		case 7:
			if (BaseLevel < 45) { mes "Super Aprendiz requer Nível de Base 45."; return; }
			jobchange Job_SuperNovice;
			break;
		case 8: jobchange Job_Taekwon; break;
		case 9: jobchange Job_Gunslinger; break;
		case 10: jobchange Job_Ninja; break;
		default: return;
		}
		specialeffect2 EF_ANGEL2;
		mes "Parabéns pela mudança de classe!";
		return;
	}

	// Aprendiz T. -> 1ª Classe Trans
	if (Class == Job_Novice_High) {
		if (JobLevel < 10) {
			mes "Você precisa de Nível de Classe 10 como Aprendiz T.";
			return;
		}
		if (lastJob) {
			jobchange roclass(eaclass(lastJob) & EAJ_BASEMASK | EAJL_UPPER);
			specialeffect2 EF_ANGEL2;
			mes "Você avançou para sua 1ª Classe Transcendental!";
			return;
		}
		mes "Escolha sua 1ª Classe Transcendental:";
		next;
		switch (select("Espadachim T.", "Mago T.", "Arqueiro T.", "Noviço T.", "Mercador T.", "Gatuno T.", "Cancelar")) {
		case 1: jobchange Job_Swordman_High; break;
		case 2: jobchange Job_Mage_High; break;
		case 3: jobchange Job_Archer_High; break;
		case 4: jobchange Job_Acolyte_High; break;
		case 5: jobchange Job_Merchant_High; break;
		case 6: jobchange Job_Thief_High; break;
		default: return;
		}
		specialeffect2 EF_ANGEL2;
		mes "Parabéns!";
		return;
	}

	// 1ª Classe -> 2ª Classe (ou Taekwon -> MTK/SL)
	if (JobLevel < 40 && (eaclass() & EAJ_UPPERMASK) == eaclass()) {
		mes "Você precisa de pelo menos Nível de Classe 40 (recomendado 50).";
		return;
	}

	switch (Class) {
	case Job_Swordman:
		.@opt = select("Cavaleiro", "Templário", "Cancelar");
		if (.@opt == 1) jobchange Job_Knight;
		else if (.@opt == 2) jobchange Job_Crusader;
		else return;
		break;
	case Job_Mage:
		.@opt = select("Bruxo", "Sábio", "Cancelar");
		if (.@opt == 1) jobchange Job_Wizard;
		else if (.@opt == 2) jobchange Job_Sage;
		else return;
		break;
	case Job_Archer:
		.@opt = select("Caçador", (Sex == SEX_MALE ? "Bardo" : "Odalisca"), "Cancelar");
		if (.@opt == 1) jobchange Job_Hunter;
		else if (.@opt == 2) jobchange (Sex == SEX_MALE ? Job_Bard : Job_Dancer);
		else return;
		break;
	case Job_Acolyte:
		.@opt = select("Sacerdote", "Monge", "Cancelar");
		if (.@opt == 1) jobchange Job_Priest;
		else if (.@opt == 2) jobchange Job_Monk;
		else return;
		break;
	case Job_Merchant:
		.@opt = select("Ferreiro", "Alquimista", "Cancelar");
		if (.@opt == 1) jobchange Job_Blacksmith;
		else if (.@opt == 2) jobchange Job_Alchemist;
		else return;
		break;
	case Job_Thief:
		.@opt = select("Mercenário", "Arruaceiro", "Cancelar");
		if (.@opt == 1) jobchange Job_Assassin;
		else if (.@opt == 2) jobchange Job_Rogue;
		else return;
		break;
	case Job_Taekwon:
		.@opt = select("Mestre Taekwon", "Espiritualista", "Cancelar");
		if (.@opt == 1) jobchange Job_Star_Gladiator;
		else if (.@opt == 2) jobchange Job_Soul_Linker;
		else return;
		break;

	// 1ª Trans -> 2ª Trans (99/70)
	case Job_Swordman_High:
		if (JobLevel < 40) { mes "Requer Nível de Classe 40+."; return; }
		.@opt = select("Lorde (Lord Knight)", "Paladino (Paladin)", "Cancelar");
		if (.@opt == 1) jobchange Job_Lord_Knight;
		else if (.@opt == 2) jobchange Job_Paladin;
		else return;
		break;
	case Job_Mage_High:
		if (JobLevel < 40) { mes "Requer Nível de Classe 40+."; return; }
		.@opt = select("Arquimago (High Wizard)", "Professor (Scholar)", "Cancelar");
		if (.@opt == 1) jobchange Job_High_Wizard;
		else if (.@opt == 2) jobchange Job_Professor;
		else return;
		break;
	case Job_Archer_High:
		if (JobLevel < 40) { mes "Requer Nível de Classe 40+."; return; }
		.@opt = select("Atirador de Elite (Sniper)", (Sex == SEX_MALE ? "Menestrel (Clown)" : "Cigana (Gypsy)"), "Cancelar");
		if (.@opt == 1) jobchange Job_Sniper;
		else if (.@opt == 2) jobchange (Sex == SEX_MALE ? Job_Clown : Job_Gypsy);
		else return;
		break;
	case Job_Acolyte_High:
		if (JobLevel < 40) { mes "Requer Nível de Classe 40+."; return; }
		.@opt = select("Sumo Sacerdote (High Priest)", "Mestre (Champion)", "Cancelar");
		if (.@opt == 1) jobchange Job_High_Priest;
		else if (.@opt == 2) jobchange Job_Champion;
		else return;
		break;
	case Job_Merchant_High:
		if (JobLevel < 40) { mes "Requer Nível de Classe 40+."; return; }
		.@opt = select("Mestre-Ferreiro (Whitesmith)", "Criador (Creator)", "Cancelar");
		if (.@opt == 1) jobchange Job_Whitesmith;
		else if (.@opt == 2) jobchange Job_Creator;
		else return;
		break;
	case Job_Thief_High:
		if (JobLevel < 40) { mes "Requer Nível de Classe 40+."; return; }
		.@opt = select("Algoz (Assassin Cross)", "Desordeiro (Stalker)", "Cancelar");
		if (.@opt == 1) jobchange Job_Assassin_Cross;
		else if (.@opt == 2) jobchange Job_Stalker;
		else return;
		break;

	// 2ª Classe Normal -> Renascer (Reborn)
	default:
		if (Upper == 0 && (eaclass() & EAJL_2)) {
			if (BaseLevel < 99 || JobLevel < 50) {
				mes "Para ^B8860BRenascer (Reborn)^000000 você precisa estar no nível ^0055FF99/50^000000.";
				return;
			}
			mes "Deseja ^B8860BRenascer^000000 para se tornar uma Transclasse (99/70)?";
			next;
			if (select("Sim, Renascer agora!", "Cancelar") == 2) return;
			lastJob = Class;
			jobchange Job_Novice_High;
			resetlvl(1);
			specialeffect2 EF_ANGEL2;
			mes "Você renasceu como Aprendiz T.! Use o atalho de nível ou upe até Job 10 para virar Transclasse.";
			return;
		}
		mes "Você já atingiu o estágio máximo Pré-Renewal para esta classe!";
		return;
	}
	specialeffect2 EF_ANGEL2;
	mes "Parabéns pela sua nova classe!";
	return;

L_PlatinumSkills:
	skill "NV_FIRSTAID", 1, SKILL_PERM;
	switch (BaseClass) {
	case Job_Novice:
		if (Class != Job_SuperNovice) skill "NV_TRICKDEAD", 1, SKILL_PERM;
		break;
	case Job_Swordman:
		skill "SM_MOVINGRECOVERY", 1, SKILL_PERM;
		skill "SM_FATALBLOW", 1, SKILL_PERM;
		skill "SM_AUTOBERSERK", 1, SKILL_PERM;
		break;
	case Job_Mage:
		skill "MG_ENERGYCOAT", 1, SKILL_PERM;
		break;
	case Job_Archer:
		skill "AC_MAKINGARROW", 1, SKILL_PERM;
		skill "AC_CHARGEARROW", 1, SKILL_PERM;
		break;
	case Job_Acolyte:
		skill "AL_HOLYLIGHT", 1, SKILL_PERM;
		break;
	case Job_Merchant:
		skill "MC_CARTREVOLUTION", 1, SKILL_PERM;
		skill "MC_CHANGECART", 1, SKILL_PERM;
		skill "MC_LOUD", 1, SKILL_PERM;
		break;
	case Job_Thief:
		skill "TF_SPRINKLESAND", 1, SKILL_PERM;
		skill "TF_BACKSLIDING", 1, SKILL_PERM;
		skill "TF_PICKSTONE", 1, SKILL_PERM;
		skill "TF_THROWSTONE", 1, SKILL_PERM;
		break;
	}
	switch (BaseJob) {
	case Job_Knight: skill "KN_CHARGEATK", 1, SKILL_PERM; break;
	case Job_Crusader: skill "CR_SHRINK", 1, SKILL_PERM; break;
	case Job_Priest: skill "PR_REDEMPTIO", 1, SKILL_PERM; break;
	case Job_Monk:
		skill "MO_KITRANSLATION", 1, SKILL_PERM;
		skill "MO_BALKYOUNG", 1, SKILL_PERM;
		break;
	case Job_Wizard: skill "WZ_SIGHTBLASTER", 1, SKILL_PERM; break;
	case Job_Sage:
		skill "SA_CREATECON", 1, SKILL_PERM;
		skill "SA_ELEMENTWATER", 1, SKILL_PERM;
		skill "SA_ELEMENTGROUND", 1, SKILL_PERM;
		skill "SA_ELEMENTFIRE", 1, SKILL_PERM;
		skill "SA_ELEMENTWIND", 1, SKILL_PERM;
		break;
	case Job_Hunter: skill "HT_PHANTASMIC", 1, SKILL_PERM; break;
	case Job_Bard: skill "BA_PANGVOICE", 1, SKILL_PERM; break;
	case Job_Dancer: skill "DC_WINKCHARM", 1, SKILL_PERM; break;
	case Job_Blacksmith:
		skill "BS_UNFAIRLYTRICK", 1, SKILL_PERM;
		skill "BS_GREED", 1, SKILL_PERM;
		break;
	case Job_Alchemist: skill "AM_BIOETHICS", 1, SKILL_PERM; break;
	case Job_Assassin:
		skill "AS_SONICACCEL", 1, SKILL_PERM;
		skill "AS_VENOMKNIFE", 1, SKILL_PERM;
		break;
	case Job_Rogue: skill "RG_CLOSECONFINE", 1, SKILL_PERM; break;
	}
	specialeffect2 EF_BEGINSPELL6;
	mes "[ ^B8860BMestra das Classes^000000 ]";
	mes "Todas as habilidades de platina da sua classe foram concedidas!";
	return;

L_Rental:
	if (ismounting()) {
		mes "Desmonte da sua montaria atual primeiro.";
		return;
	}
	if ((BaseClass == Job_Merchant || Class == Job_SuperNovice) && getskilllv("MC_PUSHCART") > 0 && !checkcart()) {
		setcart;
		mes "Carrinho entregue!";
		return;
	}
	if (BaseJob == Job_Hunter && getskilllv("HT_FALCON") > 0 && !checkfalcon()) {
		setfalcon;
		mes "Falcão entregue!";
		return;
	}
	if ((BaseJob == Job_Knight || BaseJob == Job_Crusader) && getskilllv("KN_RIDING") > 0 && !checkriding()) {
		setriding;
		mes "PecoPeco entregue!";
		return;
	}
	mes "Você já possui seu aluguel ou não tem a habilidade necessária (Usar Carrinho, Adestrar Falcão ou Montaria).";
	return;

OnInit:
	waitingroom "[ Classes 99/70 ]", 0;
	end;
}
